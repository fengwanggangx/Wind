cmake_minimum_required(VERSION 3.20)
project(Wind LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
 set(WIND_DEPS "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Wind dependency bundle")
 set(WIND_MODE "debug" CACHE STRING "Dependency configuration: debug or release")
 set(depInc "${WIND_DEPS}/include/${WIND_MODE}")
 set(depLib "${WIND_DEPS}/libs/${WIND_MODE}")
 file(GLOB supportSources CONFIGURE_DEPENDS basic/*.cpp common/*.cpp database/*.cpp ini/*.cpp network/*.cpp thread/*.cpp)
 file(GLOB protobufSources CONFIGURE_DEPENDS request/*.cc request/v1/*.cc)
 file(GLOB abslLibraries "${depLib}/libabsl_*.so")
 add_library(wind_support STATIC ${supportSources} ${protobufSources} request/request.cpp request/RequestCenter.cpp request/MarketTypes.cpp system/CSession.cpp system/CHostMgr.cpp)
 target_include_directories(wind_support PUBLIC "${depInc}/protobuf" "${depInc}/libevent" "${depInc}/mysql" "${depInc}/mysql/mysql" "${depInc}/sqlite" "${depInc}/rapidjson" "${depInc}/ini" request)
 target_link_directories(wind_support PUBLIC "${depLib}")
 set(protoSuffix "")
 if(WIND_MODE STREQUAL "debug")
  set(protoSuffix "d")
 endif()
 target_link_libraries(wind_support PUBLIC protobuf${protoSuffix} ${abslLibraries} utf8_validity utf8_range mysqlclient sqlite3 event event_core event_extra event_openssl event_pthreads Threads::Threads)

file(GLOB strategySources CONFIGURE_DEPENDS strategy/*.cpp strategy/strategies/*.cpp)
file(GLOB tradingSources CONFIGURE_DEPENDS trading/*.cpp)
add_library(wind_trading STATIC ${tradingSources})
target_link_libraries(wind_trading PUBLIC wind_support)
add_executable(Wind main.cpp system/CBootLoader.cpp system/CBrokerService.cpp system/CSubscriptionMgr.cpp ${strategySources})
target_link_libraries(Wind PRIVATE wind_trading)
add_executable(trading_protocol_tests trading/tests/TradeProtocolTests.cpp)
target_link_libraries(trading_protocol_tests PRIVATE wind_trading)
target_include_directories(trading_protocol_tests PRIVATE trading)
add_test(NAME trading_protocol_tests COMMAND trading_protocol_tests)

set(CMAKE_BUILD_RPATH "${depLib}")

add_executable(broker_client_probe trading/tests/BrokerClientProbe.cpp)
target_link_libraries(broker_client_probe PRIVATE wind_trading)
