cmake_minimum_required(VERSION 3.20)
project(WindSimulatedCounter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(trading_counter TradeTypes.cpp CSimulatedCounter.cpp CInProcessGateway.cpp)
target_include_directories(trading_counter PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(trading_counter PUBLIC Threads::Threads)
if(MSVC)
 target_compile_options(trading_counter PRIVATE /utf-8)
endif()
enable_testing()
add_executable(counter_tests tests/CounterTests.cpp)
target_link_libraries(counter_tests PRIVATE trading_counter)
add_test(NAME counter_tests COMMAND counter_tests)
option(COUNTER_MYSQL_TESTS "Build isolated MySQL integration tests" OFF)
if(COUNTER_MYSQL_TESTS)
 find_path(MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql)
 find_library(MYSQL_CLIENT mysqlclient REQUIRED)
 find_package(SQLite3 REQUIRED)
 add_executable(counter_mysql_tests tests/MySqlCounterTests.cpp CStorager.cpp
  ../database/CDBEngine.cpp ../database/CODBC.cpp ../database/CMySQL.cpp
  ../database/CSQLite3.cpp ../database/COracle.cpp ../database/IDataBase.cpp ../database/common_db.cpp ../common/utility.cpp)
 target_include_directories(counter_mysql_tests PRIVATE ${MYSQL_INCLUDE_DIR} ${MYSQL_INCLUDE_DIR}/..)
 target_link_libraries(counter_mysql_tests PRIVATE trading_counter ${MYSQL_CLIENT} SQLite::SQLite3)
 add_test(NAME counter_mysql_tests COMMAND counter_mysql_tests)
 set_tests_properties(counter_mysql_tests PROPERTIES SKIP_RETURN_CODE 77)
endif()

option(COUNTER_PROTOCOL_TESTS "Build Wind event conversion tests with the matching Protobuf runtime" OFF)
if(COUNTER_PROTOCOL_TESTS)
 find_path(PROTOBUF_INCLUDE_DIR google/protobuf/arena.h REQUIRED)
 find_path(RAPIDJSON_INCLUDE_DIR rapidjson/document.h REQUIRED)
 find_library(PROTOBUF_LIBRARY NAMES protobufd protobuf REQUIRED)
 add_executable(counter_protocol_tests tests/TradeProtocolTests.cpp TradeProtocol.cpp
  ../request/request.cpp ../request/request.pb.cc ../request/v1/market.pb.cc ../request/MarketTypes.cpp)
 target_include_directories(counter_protocol_tests PRIVATE ${PROTOBUF_INCLUDE_DIR} ${RAPIDJSON_INCLUDE_DIR} ../request)
 set(PROTOBUF_SUPPORT_LIBRARIES "" CACHE STRING "Additional libraries required by the matching Protobuf runtime, e.g. Abseil")
 target_link_libraries(counter_protocol_tests PRIVATE trading_counter ${PROTOBUF_LIBRARY} ${PROTOBUF_SUPPORT_LIBRARIES})
 add_test(NAME counter_protocol_tests COMMAND counter_protocol_tests)
endif()
